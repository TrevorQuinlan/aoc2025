def importRangesAndIDs(filepath):
    with open(f"{filepath}", "r") as f:
        while(True):
            line = f.readline()
            if "-" in line:
                values = line.strip("\n").split("-")
                ranges.append(values)
                continue
            if line == "\n":
                continue
            if not line:
                break
            else:
                ids.append(line.strip())
